# nxnativeios: SPIRV-Cross (the commit nxnative's dm/gfx/tools/spirv_cross_bin/build.sh pins) as a shared library with the CLI's entry point, spirv_cross_main.
# -DSPVC_SRC=<SPIRV-Cross checkout>. iOS: -DCMAKE_SYSTEM_NAME=iOS -DCMAKE_OSX_DEPLOYMENT_TARGET=26.0
cmake_minimum_required(VERSION 3.20)
project(SpirvCross CXX)
set(CMAKE_CXX_STANDARD 11)
set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_SKIP_INSTALL ON CACHE BOOL "" FORCE)
set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_SHARED OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(${SPVC_SRC} spvc EXCLUDE_FROM_ALL)
# main.cpp's own exit() calls become nx_spvc_exit() (wrapper.cpp: it throws) in a copy; a -Dexit= macro would also rename the C library's exit in the system headers
file(READ ${SPVC_SRC}/main.cpp CLI)
string(REGEX REPLACE "([^_A-Za-z0-9])exit\\(" "\\1nx_spvc_exit(" CLI "${CLI}")
string(REGEX MATCHALL "nx_spvc_exit\\(" HITS "${CLI}")
list(LENGTH HITS NHITS)
if(NHITS LESS 1)
  message(FATAL_ERROR "no exit() call found in main.cpp: SPIRV-Cross changed")
endif()
file(WRITE ${CMAKE_BINARY_DIR}/spvc_cli.cpp "extern \"C\" [[noreturn]] void nx_spvc_exit(int code);\n#line 1 \"main.cpp\"\n${CLI}")
message(STATUS "SPIRV-Cross CLI: ${NHITS} exit() calls redirected")
add_library(SpirvCross SHARED ${CMAKE_BINARY_DIR}/spvc_cli.cpp wrapper.cpp)
set_source_files_properties(${CMAKE_BINARY_DIR}/spvc_cli.cpp PROPERTIES COMPILE_DEFINITIONS "main=spirv_cross_cli_main")
target_include_directories(SpirvCross PRIVATE ${SPVC_SRC})
target_link_libraries(SpirvCross PRIVATE spirv-cross-glsl spirv-cross-hlsl spirv-cross-cpp spirv-cross-reflect spirv-cross-msl spirv-cross-util spirv-cross-core)
set_target_properties(SpirvCross PROPERTIES CXX_VISIBILITY_PRESET hidden)
